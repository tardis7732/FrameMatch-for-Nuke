cmake_minimum_required(VERSION 3.20)
project(FrameMatchOFX LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(FrameMatch SHARED src/plugin.cpp src/matcher.cpp)
target_include_directories(FrameMatch PRIVATE vendor/openfx/include src)
target_compile_definitions(FrameMatch PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set(FRAMEMATCH_PLUGIN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/ofx" CACHE PATH "OFX bundle output root")
set_target_properties(FrameMatch PROPERTIES PREFIX "" SUFFIX ".ofx"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${FRAMEMATCH_PLUGIN_ROOT}/FrameMatch.ofx.bundle/Contents/Win64")
if(MSVC)
  target_compile_options(FrameMatch PRIVATE /O2 /W4 /EHsc)
endif()
add_executable(matcher_tests tests/matcher_tests.cpp src/matcher.cpp)
target_include_directories(matcher_tests PRIVATE src)
if(MSVC)
  target_compile_options(matcher_tests PRIVATE /O2 /W4 /EHsc)
endif()
enable_testing()
add_test(NAME matcher_tests COMMAND matcher_tests)
add_executable(render_tests tests/render_tests.cpp src/matcher.cpp)
target_include_directories(render_tests PRIVATE vendor/openfx/include src)
target_compile_definitions(render_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME render_tests COMMAND render_tests)
